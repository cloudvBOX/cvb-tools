cloudvBOX public repository

This page hosts APT/YUM artifacts and public keys for cvb-tools.

APT

curl -fsSL https://repos.cloudvbox.com/cvb-tools.gpg | sudo tee /usr/share/keyrings/cvb-tools.gpg >/dev/null
echo "deb [signed-by=/usr/share/keyrings/cvb-tools.gpg] https://repos.cloudvbox.com/apt trixie main" | sudo tee /etc/apt/sources.list.d/cvb-tools.list
sudo apt update && sudo apt install cvb-tools

YUM/DNF

sudo tee /etc/yum.repos.d/cvb-tools.repo <<'EOF'
[cvb-tools]
name=cvb-tools
baseurl=https://repos.cloudvbox.com/yum
enabled=1
gpgcheck=0
EOF
sudo dnf makecache && sudo dnf install cvb-tools
